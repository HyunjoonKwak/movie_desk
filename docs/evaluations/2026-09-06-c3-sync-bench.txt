import "fake-indexeddb/auto";
import { it } from "vitest";
import { useProjectStore } from "@/stores/project-store";
import { appendFunnelRow, discardFunnelQueue, trimFunnelRows, type FunnelRow } from "@/persistence/funnel-log";
import { observeFunnel, recordFunnel, setFunnelEnabled, settleFunnelCollection } from "../collector";
import { computeFunnel } from "../compute";
it("measures synchronous logging work with 1000 assets", async () => {
  const initial = useProjectStore.getState();
  const state = { ...initial, project: { ...initial.project, mediaLibrary: Array.from({ length: 1000 }, () => ({ id: "synthetic" })) } } as unknown as typeof initial;
  let listener: Parameters<Parameters<typeof observeFunnel>[1]>[0] = () => {};
  setFunnelEnabled(true);
  observeFunnel(state, (fn) => { listener = fn; return () => {}; }, recordFunnel);
  let maxObserveMs = 0;
  let maxEnqueueMs = 0;
  for (let n = 0; n < 1000; n++) {
    const next = { ...state, history: { past: [{ label: "synthetic", before: state.project, after: state.project, at: n }], future: [] } };
    let start = performance.now();
    listener(next, state);
    maxObserveMs = Math.max(maxObserveMs, performance.now() - start);
    const row = { id: crypto.randomUUID(), projectId: "a".repeat(64), at: n, event: "command", data: {} };
    start = performance.now();
    appendFunnelRow(row);
    maxEnqueueMs = Math.max(maxEnqueueMs, performance.now() - start);
  }
  setFunnelEnabled(false);
  discardFunnelQueue();
  await settleFunnelCollection();
  const rows = Array.from({ length: 5000 }, (_, n) => ({ id: crypto.randomUUID(), projectId: String(n % 5), at: n, event: "command", data: {} })) as FunnelRow[];
  let start = performance.now(); trimFunnelRows(rows); const trim5000Ms = performance.now() - start;
  start = performance.now(); computeFunnel(rows); const compute5000Ms = performance.now() - start;
  console.log(JSON.stringify({ assets: 1000, calls: 1000, maxObserveMs, maxEnqueueMs, trim5000Ms, compute5000Ms }));
});
