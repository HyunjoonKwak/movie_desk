import { writeFileSync } from "node:fs";
import { createServer } from "node:http";
import { createRequire } from "node:module";
const { reentryBundle } = await import(`${process.cwd()}/scripts/color/reentry-bundle.mjs`);

const require = createRequire(`${process.cwd()}/apps/web/package.json`);
const { chromium } = require("@playwright/test");
const baseline = "84a26767d60bb58ddcdb05861492abacb41b13c9";
const server = createServer((_, response) =>
  response.end("<!doctype html><title>Paired GPU</title>"),
);
await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
const browser = await chromium.launch({ args: ["--use-angle=metal"] });
try {
  const page = await browser.newPage();
  await page.goto(`http://127.0.0.1:${server.address().port}`);
  await page.addScriptTag({ content: reentryBundle(baseline) });
  await page.evaluate(() => {
    window.__pairedBase = window.__reentry;
  });
  await page.addScriptTag({ content: reentryBundle() });
  const result = await page.evaluate(async () => {
    const before = window.__pairedBase;
    const after = window.__reentry;
    const implementations = [before, after];
    const source = (width, height) => {
      const canvas = document.createElement("canvas");
      canvas.width = width;
      canvas.height = height;
      const ctx = canvas.getContext("2d");
      ctx.fillStyle = "#808080";
      ctx.fillRect(0, 0, width, height);
      return canvas;
    };
    const large = source(3840, 2160);
    const videos = [
      new VideoFrame(large, { timestamp: 0 }),
      new VideoFrame(new Uint8Array(1920 * 1080 * 3 / 2).fill(128), {
        format: "I420", codedWidth: 1920, codedHeight: 1080, timestamp: 0,
        colorSpace: { primaries: "bt709", transfer: "bt709", matrix: "bt709", fullRange: false },
      }),
      new VideoFrame(source(6000, 4000), { timestamp: 0 }),
    ];
    const videoAssets = [
      { id: "4k", kind: "video", width: 3840, height: 2160 },
      { id: "1080p", kind: "video", width: 1920, height: 1080 },
      { id: "oversized", kind: "video", width: 6000, height: 4000 },
    ];
    const stills = await Promise.all([0, 1, 2].map(() => createImageBitmap(source(1920, 1080))));
    const stillAssets = stills.map((_, i) => ({
      id: `still-${i}`,
      kind: "image",
      width: 1920,
      height: 1080,
    }));
    for (const implementation of implementations) {
      videos.forEach((frame, i) => implementation.decoded.set(videoAssets[i].id, frame));
      stills.forEach((bitmap, i) => implementation.sources.set(stillAssets[i].id, bitmap));
    }
    const media = (asset) => ({
      id: asset.id,
      kind: "media",
      assetId: asset.id,
      start: 0,
      duration: 1000,
      trimIn: 0,
      speed: 1,
      keyframes: [],
      effects: [],
    });
    const title = {
      id: "title",
      kind: "text",
      text: "4K video + 1080p title",
      font: "sans-serif",
      size: 80,
      color: "#ffffff",
      shadow: false,
      start: 0,
      duration: 1000,
      keyframes: [],
      effects: [],
    };
    const cases = [
      ["opaque-native-video-bypass", [videoAssets[1]], [media(videoAssets[1])]],
      ["opaque-native-still-bypass", [stillAssets[0]], [media(stillAssets[0])]],
    ];
    const median = (values) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
    const results = [];
    for (const [name, assets, clips] of cases) {
      const project = {
        id: "paired",
        resolution: { w: 1920, h: 1080 },
        mediaLibrary: assets,
        timeline: {
          playhead: 0,
          duration: 1000,
          tracks: [...clips]
            .reverse()
            .map((clip, i) => ({ id: `track-${i}`, kind: "video", clips: [clip] })),
        },
      };
      const instances = implementations.map((implementation) => {
        const canvas = document.createElement("canvas");
        canvas.width = 1920;
        canvas.height = 1080;
        const compositor = new implementation.Compositor(canvas);
        const gl = canvas.getContext("webgl2");
        const counters = { targetAllocations: 0, textureDeletes: 0, fboCreates: 0, fboDeletes: 0 };
        const image = gl.texImage2D.bind(gl);
        const remove = gl.deleteTexture.bind(gl);
        const createFbo = gl.createFramebuffer.bind(gl);
        const removeFbo = gl.deleteFramebuffer.bind(gl);
        gl.texImage2D = (...args) => {
          if (args.length === 9 && args[3] > 2 && args[4] > 2) counters.targetAllocations++;
          return image(...args);
        };
        gl.deleteTexture = (texture) => {
          counters.textureDeletes++;
          return remove(texture);
        };
        gl.createFramebuffer = () => {
          counters.fboCreates++;
          return createFbo();
        };
        gl.deleteFramebuffer = (fbo) => {
          counters.fboDeletes++;
          return removeFbo(fbo);
        };
        const pixel = new Uint8Array(4);
        const render = async () => {
          await compositor.renderFrame(project, (id) => assets.find((asset) => asset.id === id));
          gl.finish();
          gl.readPixels(0, 0, 1, 1, gl.RGBA, gl.UNSIGNED_BYTE, pixel);
          if (gl.isContextLost() || gl.getError() || pixel[0] < 120 || pixel[0] > 136)
            throw new Error(`${name}: invalid completed frame ${pixel}`);
        };
        return { compositor, gl, counters, render };
      });
      try {
        for (let warmup = 0; warmup < 30; warmup++) {
          for (const instance of instances) await instance.render();
        }
        const pixels = instances.map(({ gl }) => {
          const row = new Uint8Array(1920 * 4);
          gl.readPixels(0, 540, 1920, 1, gl.RGBA, gl.UNSIGNED_BYTE, row);
          return row;
        });
        if (!pixels[0].every((v, i) => v === pixels[1][i]))
          throw new Error(`${name}: paired pixels differ`);
        for (const { counters } of instances)
          for (const key of Object.keys(counters)) counters[key] = 0;
        const batchMeans = [[], []];
        const samples = [[], []];
        for (let batch = 0; batch < 10; batch++) {
          for (const index of batch % 2 ? [1, 0] : [0, 1]) {
            const start = performance.now();
            for (let frame = 0; frame < 30; frame++) {
              const at = performance.now();
              await instances[index].render();
              samples[index].push(performance.now() - at);
            }
            batchMeans[index].push((performance.now() - start) / 30);
          }
        }
        const rows = instances.map((instance, i) => {
          const sorted = [...samples[i]].sort((a, b) => a - b);
          const counters = { ...instance.counters };
          if (Object.values(counters).some((value) => value !== 0))
            throw new Error(`${name}: steady-state reallocation ${JSON.stringify(counters)}`);
          return {
            implementation: i === 0 ? "baseline" : "modified",
            frames: sorted.length,
            p50Ms: median(sorted),
            p95Ms: sorted[Math.floor(sorted.length * 0.95)],
            medianBatchMeanMs: median(batchMeans[i]),
            batchMeans: batchMeans[i],
            blackFrames: 0,
            counters,
          };
        });
        const pairedRatios = batchMeans[0].map((value, i) => batchMeans[1][i] / value);
        results.push({ name, rows, pairedRatios, medianPairedRatio: median(pairedRatios) });
      } finally {
        for (const instance of instances) instance.compositor.dispose();
      }
    }
    for (const frame of videos) frame.close();
    for (const bitmap of stills) bitmap.close();
    return results;
  });
  const report = {
    baseline,
    protocol:
      "Same page, two warmed contexts; 30 alternating warmups, then 10 paired 30-frame batches with reversed implementation order each batch; every timed frame forces 1-pixel readback and checks neutral output, GL errors and context health.",
    results: result,
  };
  writeFileSync(
    `${process.cwd()}/docs/evaluations/2026-09-07-b5-phase3-bypass.json`,
    `${JSON.stringify(report, null, 2)}\n`,
  );
  // biome-ignore lint/suspicious/noConsole: CLI measurement.
  console.log(JSON.stringify(report, null, 2));
} finally {
  await browser.close();
  await new Promise((resolve) => server.close(resolve));
}
